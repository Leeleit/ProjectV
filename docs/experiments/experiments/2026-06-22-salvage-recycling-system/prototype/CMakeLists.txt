cmake_minimum_required(VERSION 3.28)
project(salvage_bench LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 26)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_CXX_FLAGS_RELEASE "-O3 -march=native -DNDEBUG -Wall -Wextra -Wpedantic")
add_executable(salvage_bench salvage_bench.cpp)
find_package(Threads REQUIRED)
target_link_libraries(salvage_bench PRIVATE Threads::Threads)
