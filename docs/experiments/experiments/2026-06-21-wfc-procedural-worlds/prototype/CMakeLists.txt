cmake_minimum_required(VERSION 3.28)
project(wfc_bench CXX)
set(CMAKE_CXX_STANDARD 26)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_CXX_EXTENSIONS OFF)
if(NOT CMAKE_BUILD_TYPE)
    set(CMAKE_BUILD_TYPE Release CACHE STRING "" FORCE)
endif()
add_compile_options(-O3 -march=native -DNDEBUG -Wall -Wextra -Wno-unused-parameter)
add_executable(wfc_bench bench.cpp)
target_include_directories(wfc_bench PRIVATE ${CMAKE_CURRENT_SOURCE_DIR})
